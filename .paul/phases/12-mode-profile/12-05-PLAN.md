<objective>
## Goal
Add a "LOCAL LEADERBOARD & UPLOAD" button to the Profile & Settings screen. Tapping it opens a new screen showing how many session scores are stored locally, how many slots remain before the 40-session cap is reached, and an UPLOAD SCORES button. Uploading requires watching an ad — after the ad completes, the local sessions are cleared and their data appears as a score history section below the upload button area in Profile & Settings.

## Purpose
Supabase score tracking (Phase 11-02) handles individual question-level persistence for logged-in users. This plan adds a parallel local session log that works for both guests and logged-in users — capped at 40 sessions, gated behind an ad to upload. The cap creates natural monetisation pressure; the ad gate makes upload the rewarded action rather than just completion.

## Output
- `bp_local_sessions` localStorage key — JSON array, max 40 session objects
- `saveLocalSession(sessionObj)` in app.js — called from showSummary for all users
- `#screen-local-leaderboard` — new screen (header back to Profile & Settings, session count, slot counter, UPLOAD SCORES button)
- After upload ad: local sessions cleared, uploaded history rendered below the LOCAL LEADERBOARD & UPLOAD button in `#screen-profile`
- "LOCAL LEADERBOARD & UPLOAD" button added to `#screen-profile` (logged-in section)
- CSS for new screen and history section
</objective>

    **saveLocalSession(sessionObj)** — called from showSummary for all users:
    ```js
    function saveLocalSession(sessionObj) {
      var sessions = _getLocalSessions();
      if (sessions.length >= LOCAL_SESSION_CAP) {
        // Cap reached — do not add. Cap warning is shown separately.
        return;
      }
      sessions.push({
        exercise_id:     sessionObj.exercise_id,
        mode:            sessionObj.mode,
        difficulty:      sessionObj.difficulty,
        correct_count:   sessionObj.correct_count,
        total_questions: sessionObj.total_questions,
        total_ms:        sessionObj.total_ms,
        date:            new Date().toISOString().slice(0, 10)
      });
      localStorage.setItem(LOCAL_SESSION_KEY, JSON.stringify(sessions));
    }
    ```

    ```js
      var sessions = _getLocalSessions();
      var count = sessions.length;
      var remaining = LOCAL_SESSION_CAP - count;

      document.getElementById('ll-count-value').textContent = count;
      document.getElementById('ll-slots-label').textContent =
        count + ' OF ' + LOCAL_SESSION_CAP + ' SLOTS USED — ' + remaining + ' REMAINING';

      // Disable upload if nothing to upload
      var btn = document.getElementById('ll-upload-btn');
      btn.disabled = count === 0;

      showScreen('local-leaderboard');
    };
    ```

    **AppState.uploadLocalScores():**
    ```js
    AppState.uploadLocalScores = function() {
      var sessions = _getLocalSessions();
      if (!sessions.length) return;

      // Show ad; after ad completes, perform the upload
      AppState._adProceedFn = function() {
        // Mark sessions as uploaded — render them in profile, then clear local storage
        AppState._renderUploadedScores(sessions);
        localStorage.setItem(LOCAL_SESSION_KEY, '[]');
        AppState.showProfile();
      };
      AppState._showAd();
    };
    ```

    **AppState._renderUploadedScores(sessions):**
    ```js
    AppState._renderUploadedScores = function(sessions) {
      var container = document.getElementById('profile-uploaded-scores');
      if (!container) return;
      container.innerHTML = '';
      container.classList.remove('hidden');

      var label = document.createElement('div');
      label.className = 'profile-uploaded-label';
      label.textContent = 'UPLOADED SESSIONS';
      container.appendChild(label);

      var table = document.createElement('table');
      table.className = 'profile-score-table';
      table.innerHTML =
        '<thead><tr><th>DRILL</th><th>SCORE</th><th>DIFF</th><th>DATE</th></tr></thead>';
      var tbody = document.createElement('tbody');

      sessions.forEach(function(s) {
        var tr = document.createElement('tr');
        tr.innerHTML =
          '<td>' + s.exercise_id.toUpperCase() + '</td>' +
          '<td>' + s.correct_count + '/' + s.total_questions + '</td>' +
          '<td>' + s.difficulty.toUpperCase().slice(0,1) + '</td>' +
          '<td>' + (s.date || '—') + '</td>';
        tbody.appendChild(tr);
      });

      table.appendChild(tbody);
      container.appendChild(table);

      // Persist the uploaded history across refreshes
      localStorage.setItem('bp_uploaded_sessions', JSON.stringify(sessions));
    };
    ```

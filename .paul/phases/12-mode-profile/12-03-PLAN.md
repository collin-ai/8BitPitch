    **app.js** — add `AppState.showDrillScores(exerciseId)`:
    ```js
    AppState.showDrillScores = async function(exerciseId) {
      var allScores = await window.BitPitch.Supabase.getScores();
      var drillScores = allScores.filter(function(s) { return s.exercise_id === exerciseId; });

      // Title
      document.getElementById('drill-scores-title').textContent =
        exerciseId.toUpperCase() + ' SCORES';

      // Level badge
      var level = window.BitPitch.Levels
        ? window.BitPitch.Levels.getLevelForExercise(exerciseId)
        : 0;
      document.getElementById('drill-scores-level-badge').textContent =
        'CURRENT LEVEL: ' + level;

      // Table
      var tbody = document.getElementById('drill-scores-body');
      var emptyEl = document.getElementById('drill-scores-empty');
      tbody.innerHTML = '';

      if (!drillScores.length) {
        emptyEl.classList.remove('hidden');
      } else {
        emptyEl.classList.add('hidden');
        drillScores.forEach(function(s) {
          var date = s.created_at ? s.created_at.slice(0,10) : '—';
          var tr = document.createElement('tr');
          tr.innerHTML =
            '<td>' + s.correct_count + '/' + s.total_questions + '</td>' +
            '<td>' + s.difficulty.toUpperCase().slice(0,1) + '</td>' +
            '<td>' + date + '</td>';
          tbody.appendChild(tr);
        });
      }

      showScreen('profile-drill');
    };
    ```

    ```css
      font-size: 8px;

      letter-spacing: 2px;
      margin-bottom: 12px;
    }
    ```

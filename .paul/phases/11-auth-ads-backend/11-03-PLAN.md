    getScores: async function() {
      var client = getClient();
      var session = await client.auth.getSession();
      var user = session.data.session ? session.data.session.user : null;
      if (!user) return [];

      var result = await client
        .from('scores')
        .select('exercise_id, mode, difficulty, correct_count, total_questions, created_at')
        .order('created_at', { ascending: false });

      if (result.error) {
        console.error('[BitPitch] getScores failed:', result.error.message);
        return [];
      }
      return result.data || [];
    }

    **renderProfile()** — async function called each time the Profile screen is opened:

    ```js
    async function renderProfile() {
      var user = AppState.currentUser;
      var guestEl = document.getElementById('profile-guest');
      var loggedEl = document.getElementById('profile-loggedin');

      if (!user) {
        guestEl.classList.remove('hidden');
        loggedEl.classList.add('hidden');
        return;
      }

      // Logged-in state
      guestEl.classList.add('hidden');
      loggedEl.classList.remove('hidden');

      // Set username (strip @bitpitch.local from email)
      var username = user.email ? user.email.replace('@bitpitch.local','') : 'USER';
      document.getElementById('profile-username').textContent = username;

      // Fetch scores
      var scores = await window.BitPitch.Supabase.getScores();
      var tbody = document.getElementById('profile-score-body');
      var noScoresEl = document.getElementById('profile-no-scores');
      tbody.innerHTML = '';

      if (!scores || scores.length === 0) {
        noScoresEl.classList.remove('hidden');
        return;
      }
      noScoresEl.classList.add('hidden');

      // Compute best correct_count per exercise
      var best = {};
      scores.forEach(function(s) {
        if (!best[s.exercise_id] || s.correct_count > best[s.exercise_id]) {
          best[s.exercise_id] = s.correct_count;
        }
      });

      // Render one row per score entry (most recent first — already ordered by API)
      // Deduplicate to show latest per exercise first, then older entries
      scores.forEach(function(s) {
        var date = s.created_at ? s.created_at.slice(0,10) : '—';
        var tr = document.createElement('tr');
        tr.innerHTML =
          '<td>' + s.exercise_id.toUpperCase() + '</td>' +
          '<td>' + s.correct_count + '/' + s.total_questions + '</td>' +
          '<td>' + best[s.exercise_id] + '</td>' +
          '<td>' + s.difficulty.toUpperCase().slice(0,1) + '</td>' +
          '<td>' + date + '</td>';
        tbody.appendChild(tr);
      });
    }
    ```
